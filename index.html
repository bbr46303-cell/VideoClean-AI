let timerInterval=null; let elapsed=0; let previewURL=null; function formatTime(seconds){ if(!isFinite(seconds)) return "00:00"; seconds=Math.floor(seconds); return String(Math.floor(seconds/60)).padStart(2,"0")+":"+String(seconds%60).padStart(2,"0"); } function showFileName(){ const f=document.getElementById("videoFile").files[0]; document.getElementById("fileName").innerText= f ? f.name : "No file selected"; if(f){ if(previewURL){ URL.revokeObjectURL(previewURL); } previewURL=URL.createObjectURL(f); const video=document.getElementById("videoPreview"); video.src=previewURL; document.getElementById("previewWrap").style.display="block"; video.onloadedmetadata=function(){ document.getElementById("previewDuration").innerText= formatTime(video.duration); }; } } function startProgress(){ elapsed=0; document.getElementById("progressBox").style.display="block"; document.getElementById("progressBar").style.width="0%"; document.getElementById("progressPercent").innerText="0%"; document.getElementById("timer").innerText="00:00"; timerInterval=setInterval(function(){ elapsed++; document.getElementById("timer").innerText= formatTime(elapsed); let percent=Math.min( 95, Math.round((elapsed/60)*95) ); document.getElementById("progressBar").style.width= percent+"%"; document.getElementById("progressPercent").innerText= percent+"%"; if(elapsed < 60){ document.getElementById("progressText").innerText= "AI processing... "+(60-elapsed)+"s remaining"; }else{ document.getElementById("progressText").innerText= "Finalizing video..."; } },1000); } function finishProgress(){ if(timerInterval){ clearInterval(timerInterval); timerInterval=null; } document.getElementById("progressBar").style.width="100%"; document.getElementById("progressPercent").innerText="100%"; document.getElementById("progressText").innerText= "Processing complete ✓"; } async function processVideo(){ const file=document.getElementById("videoFile").files[0]; const status=document.getElementById("status"); const button=document.getElementById("processButton"); if(!file){ status.innerText="Please select a video first."; return; } const formData=new FormData(); formData.append("video",file); formData.append("x",0); formData.append("y",0); formData.append("w",0); formData.append("h",0); formData.append("enhance",true); formData.append("audio_clean",true); try{ button.disabled=true; status.innerText= "⏳ Processing started. Please wait..."; startProgress(); const video=document.getElementById("videoPreview"); if(video.src){ video.currentTime=0; video.play().catch(function(){}); } const response=await fetch( "https://videoclean-ai.onrender.com/process", { method:"POST", body:formData } ); const result=await response.json(); finishProgress(); if(result.error){ status.innerText="❌ "+result.error; return; } if(result.download){ status.innerHTML= '✅ Processing complete. '+ 'Download video'; }else{ status.innerText= "✅ Processing completed."; } }catch(error){ if(timerInterval){ clearInterval(timerInterval); timerInterval=null; } status.innerText= "❌ Processing failed: "+error.message; }finally{ button.disabled=false; } }
